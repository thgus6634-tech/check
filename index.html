교사 설정 & 구글 앱스 스크립트(GAS)

배포된 웹 앱 URL을 입력하면 실시간 구글 시트 저장 기능이 활성화됩니다.

구글 시트 -> [확장 프로그램] -> [Apps Script] 에 아래 코드를 복사하여 붙여넣으세요.
※ 주의: 구글 시트 첫 번째 탭('시트1')의 A1 셀에 오늘의 암호를 적어두어야 합니다. 

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // 시트1 또는 첫 번째 탭에서 A1 셀의 암호 읽기
    var sheet1 = ss.getSheetByName('시트1') || ss.getSheets()[0];
    var correctPassword = String(sheet1.getRange('A1').getValue()).trim();

    // 1. 암호 검증 요청 처리
    if (data.action === 'verify_password') {
      var userPass = String(data.password || '').trim();
      if (userPass === correctPassword) {
        return ContentService.createTextOutput(JSON.stringify({ 
          success: true, 
          message: '암호 인증 성공' 
        })).setMimeType(ContentService.MimeType.JSON);
      } else {
        return ContentService.createTextOutput(JSON.stringify({ 
          success: false, 
          message: '오늘의 암호가 올바르지 않습니다.' 
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // 2. 마음건강 출석 데이터 저장 요청 처리
    if (data.action === 'submit_attendance') {
      // 2차 암호 재검증 (보안)
      var userPass = String(data.password || '').trim();
      if (userPass !== correctPassword) {
        return ContentService.createTextOutput(JSON.stringify({ 
          success: false, 
          message: '인증 정보가 올바르지 않습니다.' 
        })).setMimeType(ContentService.MimeType.JSON);
      }

      // 오늘 날짜 탭 생성/확인 (한국 시간 기준)
      var now = new Date();
      var dateTabName = Utilities.formatDate(now, "Asia/Seoul", "M월 d일");
      var timestampStr = Utilities.formatDate(now, "Asia/Seoul", "yyyy-MM-dd HH:mm:ss");

      var targetSheet = ss.getSheetByName(dateTabName);
      if (!targetSheet) {
        targetSheet = ss.insertSheet(dateTabName);
        // 헤더 추가
        targetSheet.appendRow([
          '작성시간(KST)', 
          '학번', 
          '이름', 
          '마음온도(°C)', 
          '신체신호', 
          '감정단어', 
          '나를 다독일 방법(5글자)', 
          '자기사랑실천'
        ]);
        // 헤더 스타일링
        var headerRange = targetSheet.getRange("A1:H1");
        headerRange.setBackground("#F3E8FF");
        headerRange.setFontWeight("bold");
        headerRange.setHorizontalAlignment("center");
      }

      // 학생 제출 데이터 추가
      targetSheet.appendRow([
        timestampStr,
        data.studentId || '',
        data.name || '',
        data.temperature || '',
        data.bodySignal || '',
        data.emotion || '',
        data.comfortNote || '',
        data.selfLove || ''
      ]);

      return ContentService.createTextOutput(JSON.stringify({ 
        success: true, 
        message: '출석 기록이 성공적으로 구글 시트에 저장되었습니다.' 
      })).setMimeType(ContentService.MimeType.JSON);
    }

    return ContentService.createTextOutput(JSON.stringify({ 
      success: false, 
      message: '알 수 없는 요청 형식입니다.' 
    })).setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ 
      success: false, 
      message: '서버 오류: ' + error.toString() 
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
